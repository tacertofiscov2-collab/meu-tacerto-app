import { useEffect, useState, useId } from "react";
import { FAIXA_INFO } from "@/lib/fiscal";

/**
 * VelocÃ­metro animado com:
 * - Arco de fundo com gradiente completo em opacidade 0.2
 * - Arco preenchido com gradiente completo em opacidade 1
 * - AnimaÃ§Ã£o de entrada 0% â†’ valor real (1200ms) + pulse do ponteiro (400ms)
 * - Trava visual em 100% (arco/ponteiro), mas o nÃºmero exibido Ã© o real
 */
export default function VelocimetroAnimado({
  percentual,
  maxWidth = 220,
  numeroClasse = "text-5xl font-bold",
}) {
  const uid = useId().replace(/:/g, "");
  const gradId = `velGrad-${uid}`;

  const pVisual = Math.max(0, Math.min(100, percentual));
  const [progresso, setProgresso] = useState(0);
  const [pulse, setPulse] = useState(false);

  useEffect(() => {
    setProgresso(0);
    setPulse(false);
    const t1 = setTimeout(() => setProgresso(pVisual), 80);
    const t2 = setTimeout(() => setPulse(true), 80 + 1200);
    const t3 = setTimeout(() => setPulse(false), 80 + 1200 + 400);
    return () => {
      clearTimeout(t1);
      clearTimeout(t2);
      clearTimeout(t3);
    };
  }, [pVisual]);

  const cx = 100;
  const cy = 100;
  const r = 80;
  const arcLength = Math.PI * r;
  const filledLength = (progresso / 100) * arcLength;
  const arcPath = `M ${cx - r} ${cy} A ${r} ${r} 0 0 1 ${cx + r} ${cy}`;

  // Ponteiro desenhado estaticamente apontando pra esquerda (rad = 180Â°);
  // depois rotacionado (progresso/100)*180Â° em torno do centro.
  const needleR = 60;
  const baseHalfWidth = 7;
  const midHalfWidth = 3;
  const midR = needleR * 0.45;

  // rad=Ï€: cos=-1, sin=0; perpX=sin=0, perpY=cos=-1
  const tipX = cx - needleR;
  const tipY = cy;
  const b1x = cx;
  const b1y = cy - baseHalfWidth;
  const b2x = cx;
  const b2y = cy + baseHalfWidth;
  const mx = cx - midR;
  const my = cy;
  const m1x = mx;
  const m1y = my - midHalfWidth;
  const m2x = mx;
  const m2y = my + midHalfWidth;
  const needlePoints = `${b1x},${b1y} ${m1x},${m1y} ${tipX},${tipY} ${m2x},${m2y} ${b2x},${b2y}`;

  const rotDeg = (progresso / 100) * 180;

  return (
    <div className="w-full flex flex-col items-center">
      <svg
        viewBox="0 0 200 120"
        className="block mx-auto w-full"
        style={{ maxWidth }}
        role="img"
        aria-label={`VelocÃ­metro fiscal: ${Math.round(percentual)} por cento`}
      >
        <defs>
          <linearGradient id={gradId} x1="0%" y1="0%" x2="100%" y2="0%">
            <stop offset="0%" stopColor={FAIXA_INFO.tranquilo.cor} />
            <stop offset="50%" stopColor={FAIXA_INFO.fique_de_olho.cor} />
            <stop offset="75%" stopColor={FAIXA_INFO.atencao.cor} />
            <stop offset="90%" stopColor={FAIXA_INFO.perto_do_limite.cor} />
            <stop offset="100%" stopColor={FAIXA_INFO.estourou.cor} />
          </linearGradient>
        </defs>

        {/* Arco de fundo â€” gradiente completo com opacidade reduzida */}
        <path
          d={arcPath}
          fill="none"
          stroke={`url(#${gradId})`}
          strokeWidth={18}
          strokeLinecap="round"
          opacity={0.2}
        />

        {/* Arco preenchido â€” animado via stroke-dasharray */}
        <path
          d={arcPath}
          fill="none"
          stroke={`url(#${gradId})`}
          strokeWidth={18}
          strokeLinecap="round"
          strokeDasharray={`${filledLength} ${arcLength}`}
          style={{
            transition: "stroke-dasharray 1200ms cubic-bezier(0.4, 0, 0.2, 1)",
          }}
        />

        {/* Ponteiro (rotate) + pulse */}
        <g
          style={{
            transformOrigin: "100px 100px",
            transform: `scale(${pulse ? 1.08 : 1})`,
            transition: "transform 400ms ease-out",
          }}
        >
          <g
            style={{
              transformOrigin: "100px 100px",
              transform: `rotate(${rotDeg}deg)`,
              transition: "transform 1200ms cubic-bezier(0.4, 0, 0.2, 1)",
            }}
          >
            <polygon points={needlePoints} fill="var(--text)" />
          </g>
          <circle cx={cx} cy={cy} r={8} fill="var(--text)" />
          <circle cx={cx} cy={cy} r={3.5} fill="var(--bg)" />
        </g>
      </svg>

      <div className="mt-1 flex flex-col items-center">
        <span className={numeroClasse} style={{ color: "var(--text)" }}>
          {Math.round(percentual)}%
        </span>
      </div>
    </div>
  );
}

function saudacaoPorHora() {
  const h = new Date().getHours();
  if (h >= 5 && h < 12) return "Bom dia,";
  if (h >= 12 && h < 18) return "Boa tarde,";

}

function hexToRgba(hex, alpha) {
  const h = String(hex).replace("#", "");
  const r = parseInt(h.substring(0, 2), 16);
  const g = parseInt(h.substring(2, 4), 16);
  const b = parseInt(h.substring(4, 6), 16);
  return `rgba(${r},${g},${b},${alpha})`;
}

function CardVelocimetroCarrossel({
  rotuloPerfil, percentual, faturado, limite, mediaMensal, projecao,
}) {
  const [pagina, setPagina] = useState(0);
  const [dragPx, setDragPx] = useState(0);
  const [arrastando, setArrastando] = useState(false);

  const containerRef = useRef(null);
  const startX = useRef(0);
  const startY = useRef(0);
  const startTime = useRef(0);
  const ativo = useRef(false);
  const eixo = useRef(null);

  const chaveFaixa = faixaDoVelocimetro(percentual);
  const corFaixa = FAIXA_INFO[chaveFaixa].cor;

  function largura() {
    return containerRef.current?.clientWidth || 1;
  }

  function inicio(x, y) {
    startX.current = x;
    startY.current = y;
    startTime.current = Date.now();
    ativo.current = true;
    eixo.current = null;
    setArrastando(true);
    setDragPx(0);
  }

  function mover(x, y) {
    if (!ativo.current) return;
    const dx = x - startX.current;
    const dy = y - startY.current;

    if (eixo.current === null) {
      if (Math.abs(dx) < 6 && Math.abs(dy) < 6) return;
      eixo.current = Math.abs(dx) > Math.abs(dy) ? "x" : "y";
      if (eixo.current === "y") {
        ativo.current = false;
        setArrastando(false);
        setDragPx(0);
        return;
      }
    }

    if (eixo.current === "x") setDragPx(dx);
  }

  function fim(x) {
    if (!ativo.current) {
      setArrastando(false);
      setDragPx(0);
      eixo.current = null;
      return;
    }
    const dx = x - startX.current;
    const dt = Date.now() - startTime.current;
    const velocidade = Math.abs(dx) / Math.max(1, dt);
    const passouMeio = Math.abs(dx) > largura() * 0.3;
    const flick = velocidade > 0.4 && Math.abs(dx) > 30;

    ativo.current = false;
    eixo.current = null;
    setArrastando(false);
    setDragPx(0);

    if (passouMeio || flick) setPagina((p) => (p === 0 ? 1 : 0));
  }

  function irPara(i) {
    setDragPx(0);
    setArrastando(false);
    setPagina(i);
  }

  const naTelaB = pagina === 1;
  const bgCard = naTelaB
    ? {
        background: `linear-gradient(160deg, ${hexToRgba(corFaixa, 0.16)} 0%, ${hexToRgba(corFaixa, 0.07)} 55%, ${hexToRgba(corFaixa, 0.03)} 100%), var(--surface)`,
        border: `1px solid ${hexToRgba(corFaixa, 0.3)}`,
      }
    : {
        background:
          "linear-gradient(160deg, rgba(255,255,255,0.05) 0%, rgba(255,255,255,0.01) 55%, rgba(255,255,255,0) 100%), var(--surface)",
        border: "1px solid var(--border)",
      };

  const larg = largura();
  const basePct = pagina === 0 ? 0 : -50;
  const dragPct = larg ? (dragPx / larg) * 50 : 0;
  const translate = basePct + dragPct;

  return (
    <div
      ref={containerRef}
      data-carrossel-velocimetro
      className="relative w-full flex-1 min-h-0 rounded-3xl overflow-hidden flex flex-col"
      style={{
        ...bgCard,
        boxShadow: "0 10px 30px rgba(0,0,0,0.3), inset 0 1px 0 rgba(255,255,255,0.06)",
