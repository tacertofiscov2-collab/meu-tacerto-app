function MiniVelocimetro({ pct = 55 }) {
  const R = 46;
  const cx = 60;
  const cy = 60;
  const circ = Math.PI * R;
  const gid = "mini-velo-grad";
  const arco = `M ${cx - R} ${cy} A ${R} ${R} 0 0 1 ${cx + R} ${cy}`;

  const needleR = R * 0.75;
  const baseHalfWidth = 4;
  const midHalfWidth = 1.7;
  const midR = needleR * 0.45;

  const tipX = cx - needleR;
  const tipY = cy;
  const b1x = cx, b1y = cy - baseHalfWidth;
  const b2x = cx, b2y = cy + baseHalfWidth;
  const mx = cx - midR, my = cy;
  const m1x = mx, m1y = my - midHalfWidth;
  const m2x = mx, m2y = my + midHalfWidth;
  const needlePoints = `${b1x},${b1y} ${m1x},${m1y} ${tipX},${tipY} ${m2x},${m2y} ${b2x},${b2y}`;

  const rotDeg = (pct / 100) * 180;

  return (
    <svg width="120" height="74" viewBox="0 0 120 74" aria-hidden>
      <defs>
        <linearGradient id={gid} gradientUnits="userSpaceOnUse" x1={cx - R} y1={cy} x2={cx + R} y2={cy}>
          <stop offset="0%" stopColor="#22c55e" />
          <stop offset="45%" stopColor="#84cc16" />
          <stop offset="70%" stopColor="#eab308" />
          <stop offset="100%" stopColor="#ef4444" />
        </linearGradient>
      </defs>
      <path d={arco} fill="none" stroke={`url(#${gid})`} strokeWidth="9" strokeLinecap="round" />
      <path
        d={arco}
        fill="none"
        stroke="var(--surface-raised)"
        strokeWidth="10"
        strokeLinecap="round"
        strokeDasharray={`${circ} ${circ}`}
        strokeDashoffset={-((pct / 100) * circ)}
        opacity="0.82"
      />
      <g style={{ transformOrigin: "60px 60px", transform: `rotate(${rotDeg}deg)` }}>
        <polygon points={needlePoints} fill="var(--text)" />
      </g>
      <circle cx={cx} cy={cy} r="5.5" fill="var(--text)" />
      <circle cx={cx} cy={cy} r="2.5" fill="var(--surface)" />
    </svg>
  );
}

  const [active, setActive] = useState(0);

    const el = scrollerRef.current;
    if (!el) return;
    const onScroll = () => setActive(Math.round(el.scrollLeft / el.clientWidth));
    el.addEventListener("scroll", onScroll, { passive: true });
    return () => el.removeEventListener("scroll", onScroll);
  }, []);

  const goTo = (i) => {
    const el = scrollerRef.current;
    if (!el) return;
    el.scrollTo({ left: i * el.clientWidth, behavior: "smooth" });
